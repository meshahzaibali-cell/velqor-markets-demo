function resetBalance(){ if(!user)return; if(user.positions.length){ $("walletMsg").textContent="Close all open trades before resetting."; return; } user.balance=500; user.wallet.push({label:"Demo balance reset",amount:500}); saveUser(); render(); $("walletMsg").textContent="Balance reset to $500."; } function setTF(value,btn){ tf=value; document.querySelectorAll(".charttools button") .forEach(b=>b.classList.remove("active")); if(btn)btn.classList.add("active"); drawChart(); } function drawChart(){ const canvas=$("chart"); if(!canvas)return; const ctx=canvas.getContext("2d"); const rect=canvas.getBoundingClientRect(); const dpr=window.devicePixelRatio||1; canvas.width=rect.width*dpr; canvas.height=rect.height*dpr; ctx.setTransform(dpr,0,0,dpr,0,0); const w=rect.width,h=rect.height; ctx.clearRect(0,0,w,h); if(!candles.length){ let p=mid; for(let i=0;i<50;i++){ let o=p; let c=o+(Math.random()-.5)*5; candles.push({ o:o,c:c, hi:Math.max(o,c)+Math.random()*2, lo:Math.min(o,c)-Math.random()*2 }); p=c; } } candles[candles.length-1].c=mid; ctx.strokeStyle="#29394b"; ctx.lineWidth=1; for(let i=0;i<5;i++){ let y=i*h/4; ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(w,y); ctx.stroke(); } const data=candles.slice(-50); const prices=data.flatMap(c=>[c.hi,c.lo]); const max=Math.max(...prices)+1; const min=Math.min(...prices)-1; const range=max-min||1; const py=p=>h-((p-min)/range)*h; const cw=w/data.length; data.forEach((c,i)=>{ const x=i*cw+cw/2; const up=c.c>=c.o; const color=up?"#16c784":"#ea5353"; ctx.strokeStyle=color; ctx.fillStyle=color; ctx.lineWidth=1; ctx.beginPath(); ctx.moveTo(x,py(c.hi)); ctx.lineTo(x,py(c.lo)); ctx.stroke(); const top=Math.min(py(c.o),py(c.c)); const height=Math.max(1,Math.abs(py(c.o)-py(c.c))); ctx.fillRect(x-cw*.3,top,cw*.6,height); }); ctx.fillStyle="#d5ad58"; ctx.font="12px Arial"; ctx.fillText(num(mid),5,15); } function updatePrice(){ const change=(Math.random()-.5)*1.2; mid=Math.max(100,mid+change); bid=mid-.15; ask=mid+.15; if(candles.length){ let c=candles[candles.length-1]; c.c=mid; c.hi=Math.max(c.hi,mid); c.lo=Math.min(c.lo,mid); } if(user){ checkStops(); render(); } if(!$("trade").classList.contains("active"))return; }); } initialize();
